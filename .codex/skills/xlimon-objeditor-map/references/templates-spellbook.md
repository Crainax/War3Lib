# 魔法书模板（Aspb）

说明：Order/基础命令ID 一致时会共享；通常同一本书内保持一致，避免串到其他书。

```ini
[AT01]
_parent = "Aspb"
Buttonpos_1 = 1 -- 按钮位置 - 普通 (X)
Buttonpos_2 = 1 -- 按钮位置 - 普通 (Y)
DataA = "" -- 共享法术CD间隔
DataB = 0
DataC = 11 -- 最小法术数量
DataD = 11 -- 最大法术数量
DataE = "resumebuild" -- 基础命令ID
Hotkey = "T" -- 热键 - 普通
Name = "|cffccffcc天赋技能树2|r" -- 名字
Tip = "|cffccffcc天赋技能树2|r(|cffffcc00G|r)" -- 提示工具 - 普通
Ubertip = [=[天赋技能树2]=] -- 提示工具 - 普通 - 扩展
item = 0 -- 物品技能
race = "human" -- 种族
```
